function timedModule(name, fn) {
  return (...args) => {
    const start = performance.now();
    try {
      return fn(...args);
    } finally {
      timers[name] = (timers[name] ?? 0) + performance.now() - start;
    }
  };
}

import useSketchStudio from './hooks/useSketchStudio';
import SketchStudioLayout from './components/SketchStudioLayout';

import { MemoryRouter } from 'react-router-dom';

function reset() {
  for (const key of Object.keys(counters)) counters[key] = 0;
}

function createRandom(seed) {
  let state = seed >>> 0 || 1;
  return () => {
    state ^= state << 13;
    state >>>= 0;
    state ^= state >>> 17;
    state ^= state << 5;
    state >>>= 0;
    return state / 4294967296;
  };
}

function buildLines(count, seed = 7) {
  const random = createRandom(seed);
  const entities = [];
  for (let index = 0; index < count; index += 1) {
    const x = random() * 20000;
    const y = random() * 20000;
    const angle = random() * Math.PI;
    const length = 200 + random() * 1800;
    entities.push({
      id: `line-${index}`,
      type: 'line',
      x1: x,
      y1: y,
      x2: x + Math.cos(angle) * length,
      y2: y + Math.sin(angle) * length,
      layerId: 'default',
      visible: true,
      locked: false,
      meta: {},
    });
  }
  return entities;
}

function buildArcs(count, seed = 11) {
  const random = createRandom(seed);
  const entities = [];
  for (let index = 0; index < count; index += 1) {
    const x = random() * 20000;
    const y = random() * 20000;
    entities.push({
      id: `arc-${index}`,
      type: 'arc',
      start: { x, y },
      control: { x: x + 200, y: y + 400 },
      end: { x: x + 400, y },
      layerId: 'default',
      visible: true,
      locked: false,
      meta: {},
    });
  }
  return entities;
}

function pointer({ x = 0, y = 0, button = 0, buttons = 1 } = {}) {
  return {
    clientX: x,
    clientY: y,
    button,
    buttons,
    ctrlKey: false,
    shiftKey: false,
    metaKey: false,
    pointerId: 1,
    preventDefault() {},
    stopPropagation() {},
    currentTarget: { setPointerCapture() {}, hasPointerCapture: () => true, releasePointerCapture() {} },
  };
}

let studio = null;
let layoutRenders = 0;
function Harness() {
  studio = useSketchStudio();
  layoutRenders += 1;
  return <SketchStudioLayout {...studio} />;
}

function mount(entities, viewport = { zoom: 0.05, panX: 0, panY: 0 }) {
  render(
    <MemoryRouter>
      <ConfirmDialogProvider>
        <Harness />
      </ConfirmDialogProvider>
    </MemoryRouter>,
  );
  act(() => studio.loadTemplate({ document: { ...sampleDocument, entities }, viewport }));
  return () => studio;
}

function timed(fn) {
  const start = performance.now();
  act(fn);
  return performance.now() - start;
}

function snapshot(label, moves, ms) {
  const per = (value) => +(value / moves).toFixed(2);
  return {
    label,
    layoutRendersPerMove: per(layoutRenders),
    entityLayerRendersPerMove: per(counters.entityLayerRenders),
    entityElementsPerMove: per(counters.entityElements),
    entityShapeRendersPerMove: per(counters.entityShapeRenders),
    dimensionLayerRendersPerMove: per(counters.dimensionLayerRenders),
    normalizeDocPerMove: per(counters.normalizeDoc),
    normalizeModelPerMove: per(counters.normalizeModel),
    pairTestsPerMove: per(counters.pairTests),
    distancePerMove: per(counters.distance),
    msPerMove: +(ms / moves).toFixed(2),
  };
}

function resetAll() {
  reset();
  layoutRenders = 0;
}

  it('select tool: pointer moves over empty space, hover changes, pan frames', () => {
    const lines = buildLines(5000);
    mount(lines);
    // Warm.
    act(() => studio.onCanvasPointerMove(pointer({ x: 1100, y: 1100, buttons: 0 })));

    // Empty space (outside the drawing: x > 20000 mm at zoom 0.05 = 1000 px).
    resetAll();
    let ms = 0;
    for (let index = 0; index < 20; index += 1) {
      ms += timed(() => studio.onCanvasPointerMove(pointer({ x: 1100 + index, y: 1100, buttons: 0 })));
    }
    results.push(snapshot('select: move over empty space (5000 lines)', 20, ms));

    // Hover changes: alternate between two line midpoints.
    const a = lines[10];
    const b = lines[20];
    const mid = (line) => ({ x: ((line.x1 + line.x2) / 2) * 0.05, y: ((line.y1 + line.y2) / 2) * 0.05 });
    resetAll();
    ms = 0;
    for (let index = 0; index < 10; index += 1) {
      const target = index % 2 ? mid(a) : mid(b);
      ms += timed(() => studio.onCanvasPointerMove(pointer({ ...target, buttons: 0 })));
    }
    results.push({ ...snapshot('select: hover change A<->B (5000 lines)', 10, ms), hovered: studio.hover.hoveredId });

    // Pan frames: middle button.
    act(() => studio.onCanvasPointerDown(pointer({ x: 1100, y: 1100, button: 1 })));
    resetAll();
    ms = 0;
    for (let index = 1; index <= 10; index += 1) {
      ms += timed(() => studio.onCanvasPointerMove(pointer({ x: 1100 + index * 5, y: 1100, button: 1, buttons: 4 })));
    }
    results.push(snapshot('pan frame (5000 lines)', 10, ms));
    act(() => studio.onCanvasPointerUp(pointer({ x: 1150, y: 1100, button: 1 })));
  });

    const lines = buildLines(5000);

      act(() => studio.onCanvasPointerDown(pointer(start)));

  it('C1 raw: old all-pairs vs new findIntersectionSnapPoint', async () => {
    const { collectSnapSegmentsFromEntities, findIntersectionSnapPoint } = await import('./utils/snapUtils');
    const { getSegmentIntersectionPoint } = await vi.importActual('./utils/hitTest');
    function oldFind(segments, worldPoint, tolerance) {
      let nearest = null;
      let nearestDistance = Infinity;
      for (let i = 0; i < segments.length; i += 1) {
        for (let j = i + 1; j < segments.length; j += 1) {
          const p = getSegmentIntersectionPoint(segments[i].start, segments[i].end, segments[j].start, segments[j].end);
          if (!p) continue;
          const d = Math.hypot(worldPoint.x - p.x, worldPoint.y - p.y);
          if (d <= tolerance && d < nearestDistance) {
            nearest = p;
            nearestDistance = d;
          }
        }
      }
      return nearest;
    }
    for (const count of [3000, 10000]) {
      const segments = collectSnapSegmentsFromEntities(buildLines(count));
      const line = buildLines(count)[5];
      const cursor = { x: (line.x1 + line.x2) / 2 + 60, y: (line.y1 + line.y2) / 2 + 60 };
      const tolerance = 12 / 0.05;
      let start = performance.now();
      const oldResult = count <= 3000 ? oldFind(segments, cursor, tolerance) : 'skipped';
      const oldMs = performance.now() - start;
      reset();
      start = performance.now();
      const newResult = findIntersectionSnapPoint(segments, cursor, tolerance);
      const newMs = performance.now() - start;
      results.push({
        label: `C1 raw ${count} segments`,
        oldPairs: (count * (count - 1)) / 2,
        oldMs: +oldMs.toFixed(2),
        newPairs: counters.pairTests,
        newMs: +newMs.toFixed(2),
        same: count <= 3000 ? JSON.stringify(oldResult) === JSON.stringify(newResult && { x: newResult.x, y: newResult.y }) : 'n/a',
      });
    }
  });

    const fs = await import('node:fs');

  });
